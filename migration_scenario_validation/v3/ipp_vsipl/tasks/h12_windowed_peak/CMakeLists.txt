cmake_minimum_required(VERSION 3.16)
project(h12_windowed_peak C)
set(CMAKE_C_STANDARD 99)
set(CMAKE_C_STANDARD_REQUIRED ON)
option(USE_GOLD "Build reference" OFF)
set(VSIPL_ROOT "" CACHE PATH "VSIPL prefix")
if(USE_GOLD)
  file(GLOB CORE_SOURCES CONFIGURE_DEPENDS ${CMAKE_CURRENT_SOURCE_DIR}/gold/*.c)
else()
  file(GLOB CORE_SOURCES CONFIGURE_DEPENDS ${CMAKE_CURRENT_SOURCE_DIR}/src/*.c)
endif()
add_library(window_spectrum STATIC ${CORE_SOURCES})
target_include_directories(window_spectrum PUBLIC ${CMAKE_CURRENT_SOURCE_DIR}/include ${CMAKE_CURRENT_SOURCE_DIR}/src)
if(USE_GOLD OR VSIPL_ROOT)
  if(NOT VSIPL_ROOT)
    message(FATAL_ERROR "VSIPL_ROOT required")
  endif()
  target_include_directories(window_spectrum PUBLIC ${VSIPL_ROOT}/include)
  target_link_libraries(window_spectrum PUBLIC ${VSIPL_ROOT}/lib/libvsip.a m)
endif()
enable_testing()
add_executable(public_test test/public_test.c)
target_link_libraries(public_test window_spectrum m)
add_test(NAME public_test COMMAND public_test)
