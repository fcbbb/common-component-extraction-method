cmake_minimum_required(VERSION 3.16)
project(h13_spectral_centroid C)
set(CMAKE_C_STANDARD 99)
set(CMAKE_C_STANDARD_REQUIRED ON)
option(USE_IPP "Build IPP source" OFF)
option(USE_GOLD "Build reference" OFF)
set(IPP_ROOT "" CACHE PATH "IPP prefix")
set(VSIPL_ROOT "" CACHE PATH "VSIPL prefix")
if(USE_GOLD)
  file(GLOB CORE_SOURCES CONFIGURE_DEPENDS ${CMAKE_CURRENT_SOURCE_DIR}/gold/*.c)
else()
  file(GLOB CORE_SOURCES CONFIGURE_DEPENDS ${CMAKE_CURRENT_SOURCE_DIR}/src/*.c)
endif()
add_library(spectral_centroid STATIC ${CORE_SOURCES})
target_include_directories(spectral_centroid PUBLIC ${CMAKE_CURRENT_SOURCE_DIR}/include ${CMAKE_CURRENT_SOURCE_DIR}/src)
if(USE_IPP)
  target_include_directories(spectral_centroid PUBLIC ${IPP_ROOT}/include)
  target_link_libraries(spectral_centroid PUBLIC ${IPP_ROOT}/lib/shared/libipps.so ${IPP_ROOT}/lib/shared/libippcore.so)
endif()
if(USE_GOLD OR VSIPL_ROOT)
  if(NOT VSIPL_ROOT)
    message(FATAL_ERROR "VSIPL_ROOT required")
  endif()
  target_include_directories(spectral_centroid PUBLIC ${VSIPL_ROOT}/include)
  target_link_libraries(spectral_centroid PUBLIC ${VSIPL_ROOT}/lib/libvsip.a m)
endif()
enable_testing()
add_executable(public_test test/public_test.c)
target_link_libraries(public_test spectral_centroid m)
add_test(NAME public_test COMMAND public_test)
