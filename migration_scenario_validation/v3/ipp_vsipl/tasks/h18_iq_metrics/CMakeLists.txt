cmake_minimum_required(VERSION 3.16)
project(h18_iq_metrics C)
set(CMAKE_C_STANDARD 99)
set(CMAKE_C_STANDARD_REQUIRED ON)
option(USE_IPP "IPP" OFF)
option(USE_GOLD "GOLD" OFF)
set(IPP_ROOT "" CACHE PATH "IPP")
set(VSIPL_ROOT "" CACHE PATH "VSIPL")
if(USE_GOLD)
 file(GLOB S CONFIGURE_DEPENDS ${CMAKE_CURRENT_SOURCE_DIR}/gold/*.c)
else()
 file(GLOB S CONFIGURE_DEPENDS ${CMAKE_CURRENT_SOURCE_DIR}/src/*.c)
endif()
add_library(iq_metrics STATIC ${S})
target_include_directories(iq_metrics PUBLIC ${CMAKE_CURRENT_SOURCE_DIR}/include)
if(USE_IPP)
 target_include_directories(iq_metrics PUBLIC ${IPP_ROOT}/include)
 target_link_libraries(iq_metrics PUBLIC ${IPP_ROOT}/lib/shared/libipps.so ${IPP_ROOT}/lib/shared/libippcore.so)
endif()
if(USE_GOLD OR VSIPL_ROOT)
 target_include_directories(iq_metrics PUBLIC ${VSIPL_ROOT}/include)
 target_link_libraries(iq_metrics PUBLIC ${VSIPL_ROOT}/lib/libvsip.a m)
endif()
add_executable(public_test test/public_test.c)
target_link_libraries(public_test iq_metrics m)
